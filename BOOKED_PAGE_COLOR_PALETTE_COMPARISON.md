// Usage in Booking Card Component
const BookingCard = ({ booking }) => {
  const colors = getServiceColors(booking.serviceType);
  const service = getServiceById(booking.serviceType);
  
  return (
    <div style={{
      background: colors.bg,
      border: `1px solid ${colors.border}`,
      borderLeft: `4px solid ${colors.accent}`,
      borderRadius: '16px',
      padding: '1.5rem',
      transition: 'all 0.3s ease',
      cursor: 'pointer',
    }}
    onMouseEnter={(e) => {
      e.currentTarget.style.transform = 'translateY(-4px)';
      e.currentTarget.style.boxShadow = `0 8px 24px ${colors.border}40`;
    }}
    onMouseLeave={(e) => {
      e.currentTarget.style.transform = 'translateY(0)';
      e.currentTarget.style.boxShadow = 'none';
    }}
    >
      {/* Service Badge */}
      <div style={{
        display: 'inline-block',
        padding: '0.4rem 0.9rem',
        borderRadius: '20px',
        fontSize: '0.75rem',
        fontWeight: '600',
        textTransform: 'uppercase',
        letterSpacing: '0.1em',
        background: colors.bg,
        color: colors.label,
        border: `1px solid ${colors.border}`,
        marginBottom: '1rem',
      }}>
        {service?.name || booking.serviceType}
      </div>
      
      {/* Booking Details */}
      <div style={{ color: colors.text }}>
        <div style={{ fontSize: '1.1rem', fontWeight: '600', marginBottom: '0.5rem' }}>
          {booking.modelName || 'Model Name'}
        </div>
        <div style={{ fontSize: '0.9rem', color: '#5A3A2A', marginBottom: '0.25rem' }}>
          {new Date(booking.appointmentDate).toLocaleDateString()}
        </div>

        </div>

          Model Fee: ${booking.modelFee || 0}
        </div>
      </div>
    </div>
  );
};
